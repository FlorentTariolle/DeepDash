% The archived frames are 64 x 64 pixel maps.  PGF writes an explicit
% /Interpolate false image attribute so PDF viewers preserve their native,
% nearest-neighbour appearance instead of smoothing them.
\newcommand{\framegraphic}[1]{%
  \pgfimage[width=2.58cm,height=2.58cm,interpolate=false]{#1}%
}
\newcommand{\placeframe}[3]{%
  \coordinate (#1center) at #2;
  \begin{scope}
    \clip[rounded corners=2.5pt]
      ($(#1center)+(-1.29,-1.29)$) rectangle
      ($(#1center)+(1.29,1.29)$);
    \node[frame] (#1) at (#1center) {\framegraphic{#3}};
  \end{scope}
}
